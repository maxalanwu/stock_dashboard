## 💼 持股

### 💰 部位損益
```dataview
TABLE shares AS "股數", entry_price AS "進場價", price AS "股價", change_pct AS "漲跌%", choice(shares, regexreplace(string(round(price * shares)), "(?<=[0-9])(?=([0-9]{3})+$)", ","), null) AS "市值", choice(shares AND entry_price, regexreplace(string(round((price - entry_price) * shares)), "(?<=[0-9])(?=([0-9]{3})+$)", ","), null) AS "損益", choice(shares AND entry_price, regexreplace(string(round(net_pnl)), "(?<=[0-9])(?=([0-9]{3})+$)", ","), null) AS "賣出淨損益", choice(entry_price, round((price - entry_price) / entry_price * 100, 2), null) AS "報酬%"
FROM "10_Stocks"
WHERE status = "持有"
FLATTEN 0.001425 * this.fee_discount AS fee_rate
FLATTEN choice(industry = "ETF", 0.001, 0.003) AS tax_rate
FLATTEN max(20, floor(entry_price * shares * fee_rate)) + max(20, floor(price * shares * fee_rate)) + floor(price * shares * tax_rate) AS sell_cost
FLATTEN (price - entry_price) * shares - sell_cost AS net_pnl
SORT price * shares DESC
```

### 盈虧總計
```dataview
TABLE WITHOUT ID length(rows) AS "檔數", regexreplace(string(round(sum(rows.cost))), "(?<=[0-9])(?=([0-9]{3})+$)", ",") AS "總成本", regexreplace(string(round(sum(rows.mv))), "(?<=[0-9])(?=([0-9]{3})+$)", ",") AS "總市值", regexreplace(string(round(sum(rows.pnl))), "(?<=[0-9])(?=([0-9]{3})+$)", ",") AS "總損益", regexreplace(string(round(sum(rows.net_pnl))), "(?<=[0-9])(?=([0-9]{3})+$)", ",") AS "賣出淨損益", round(sum(rows.pnl) / sum(rows.cost) * 100, 2) AS "總報酬%"
FROM "10_Stocks"
WHERE status = "持有" AND shares AND entry_price
FLATTEN entry_price * shares AS cost
FLATTEN price * shares AS mv
FLATTEN (price - entry_price) * shares AS pnl
FLATTEN 0.001425 * this.fee_discount AS fee_rate
FLATTEN choice(industry = "ETF", 0.001, 0.003) AS tax_rate
FLATTEN max(20, floor(entry_price * shares * fee_rate)) + max(20, floor(price * shares * fee_rate)) + floor(price * shares * tax_rate) AS sell_cost
FLATTEN (price - entry_price) * shares - sell_cost AS net_pnl
GROUP BY true
```

### 🛡️ 風控
```dataview
TABLE price AS "股價", stop_loss AS "停損", choice(stop_loss, round((price - stop_loss) / price * 100, 2), null) AS "距停損%", target_price AS "停利價", choice(target_price, round((target_price - price) / price * 100, 2), null) AS "距停利%", choice(rsi > 80, "<span style='color:#e53935;font-weight:bold'>" + rsi + " 過熱</span>", choice(rsi < 20, "<span style='color:#43a047;font-weight:bold'>" + rsi + " 超賣</span>", rsi)) AS "RSI", foreign_net_5d AS "外資5日"
FROM "10_Stocks"
WHERE status = "持有"
SORT choice(stop_loss, (price - stop_loss) / price, 999) ASC
```
